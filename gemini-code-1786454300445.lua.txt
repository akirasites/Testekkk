local function updateDropdown()
    for _, child in pairs(DropdownScroll:GetChildren()) do
        if child:IsA("TextButton") then child:Destroy() end
    end

    local ySize = 0  
    for _, player in pairs(Players:GetPlayers()) do  
        local btn = Instance.new("TextButton")  
        btn.Size = UDim2.new(1, 0, 0, 30)  
        btn.BackgroundColor3 = Color3.fromRGB(20, 20, 20)  
        btn.BackgroundTransparency = 0.2  
        btn.TextColor3 = Color3.fromRGB(255, 255, 255)  
        btn.Text = player.Name .. (player == LocalPlayer and " (Você)" or "")  
        btn.Font = Enum.Font.Gotham  
        btn.TextSize = 12  
        btn.ZIndex = 16  
        btn.Parent = DropdownScroll  

        Instance.new("UICorner", btn).CornerRadius = UDim.new(0, 4)  

        btn.Activated:Connect(function()  
            selectedPlayer = player  
            DropdownBtn.Text = "Alvo: " .. player.Name  
            DropdownScroll.Visible = false  
            DropdownBackdrop.Visible = false  
        end)  
        ySize = ySize + 30  
    end  
    DropdownScroll.CanvasSize = UDim2.new(0, 0, 0, ySize)
end

local function toggleOriginalCharacterVisibility(char, isVisible)
    local transparencyValue = isVisible and 0 or 1
    for _, obj in ipairs(char:GetDescendants()) do  
        if obj:IsA("BasePart") and obj.Name ~= "HumanoidRootPart" then  
            obj.Transparency = transparencyValue  
        elseif obj:IsA("Decal") or obj:IsA("Texture") then  
            obj.Transparency = transparencyValue  
        elseif obj:IsA("Accessory") or obj:IsA("Hat") then  
            local handle = obj:FindFirstChild("Handle")  
            if handle then handle.Transparency = transparencyValue end  
        end  
    end  
    local head = char:FindFirstChild("Head")  
    if head and head:FindFirstChildOfClass("BillboardGui") then  
        head:FindFirstChildOfClass("BillboardGui").Enabled = isVisible  
    end
end

local function applyTargetAnimations(targetChar, targetId)
    local targetHum = targetChar:FindFirstChildOfClass("Humanoid")
    if not targetHum or targetHum.RigType == Enum.HumanoidRigType.R6 then return end

    local myAnimate = targetChar:FindFirstChild("Animate")
    if not myAnimate then return end

    local success, appearance = pcall(function() return Players:GetCharacterAppearanceAsync(targetId) end)  
    if success and appearance then  
        appearance.Parent = game:GetService("ReplicatedStorage")  
        local r15Anim = appearance:WaitForChild("R15Anim", 2)  
        if r15Anim then  
            for _, animFolder in ipairs(r15Anim:GetChildren()) do  
                local myFolder = myAnimate:FindFirstChild(animFolder.Name)  
                if myFolder then  
                    local customAnims = {}  
                    for _, a in ipairs(animFolder:GetChildren()) do  
                        if a:IsA("Animation") then table.insert(customAnims, a.AnimationId) end  
                    end  
                    local index = 1  
                    for _, myA in ipairs(myFolder:GetChildren()) do  
                        if myA:IsA("Animation") and customAnims[index] then  
                            myA.AnimationId = customAnims[index]  
                            index = index + 1  
                        end  
                    end  
                end  
            end  
        end
        appearance:Destroy()   
    end
end

local function internalRestore(targetPlayer)
    if not targetPlayer or not modifiedPlayers[targetPlayer] then return end
    local data = modifiedPlayers[targetPlayer]  
    local targetChar = targetPlayer.Character  
    local targetHum = targetChar and targetChar:FindFirstChildOfClass("Humanoid")  

    if data.Type == "DirectR6" then
        -- Restaura as roupas reais sem mexer na transparência/dummy
        if data.AddedClothes then
            for _, obj in ipairs(data.AddedClothes) do
                if obj and obj.Parent then obj:Destroy() end
            end
        end
        if data.BackupClothes and targetChar then
            for _, obj in ipairs(data.BackupClothes) do
                local clone = obj:Clone()
                if clone:IsA("Decal") or clone:IsA("SpecialMesh") then
                    local head = targetChar:FindFirstChild("Head")
                    if head then clone.Parent = head end
                else
                    clone.Parent = targetChar
                end
            end
        end
    else
        -- Lógica original de restaurar o Dummy
        if data.Dummy then data.Dummy:Destroy() end  
        if data.Connection then data.Connection:Disconnect() end  

        if targetChar and targetHum then  
            toggleOriginalCharacterVisibility(targetChar, true)  
        end  
    end
    
    modifiedPlayers[targetPlayer] = nil  
end

local function processCloning(mode)
    if not selectedPlayer then return end

    local targetChar = selectedPlayer.Character  
    local targetHum = targetChar and targetChar:FindFirstChildOfClass("Humanoid")  
    if not targetChar or not targetHum then return end  

    local copyUsername = UsernameInput.Text  
    if copyUsername == "" then copyUsername = selectedPlayer.Name end  

    local btn = (mode == "R15") and ApplyR15Btn or ApplyR6Btn
    local defaultText = btn.Text
    btn.Text = "Carregando..."  

    internalRestore(selectedPlayer)

    local success, targetId = pcall(function() return Players:GetUserIdFromNameAsync(copyUsername) end)  
    if not success or not targetId then  
        btn.Text = "Usuário Inválido!"  
        task.delay(3, function() btn.Text = defaultText end)  
        return  
    end  

    -------------------------------------------------------------------------
    -- NOVA LÓGICA: R6 EM JOGO R6 (APLICAÇÃO DIRETA SEM DUMMY)
    -------------------------------------------------------------------------
    if mode == "R6" and targetHum.RigType == Enum.HumanoidRigType.R6 then
        local backupClothes = {}
        local addedClothes = {}
        
        -- 1. Tira as roupas atuais e faz backup
        for _, obj in ipairs(targetChar:GetChildren()) do
            if obj:IsA("Accessory") or obj:IsA("Hat") or obj:IsA("Shirt") or obj:IsA("Pants") or obj:IsA("ShirtGraphic") or obj:IsA("CharacterMesh") or obj:IsA("BodyColors") then
                obj.Archivable = true
                table.insert(backupClothes, obj:Clone())
                obj:Destroy()
            elseif obj:IsA("Part") and obj.Name == "Head" then
                for _, child in ipairs(obj:GetChildren()) do
                    if child:IsA("Decal") or child:IsA("SpecialMesh") then
                        child.Archivable = true
                        table.insert(backupClothes, child:Clone())
                        child:Destroy()
                    end
                end
            end
        end

        -- 2. Pega e adiciona as roupas novas
        local successApp, appearance = pcall(function() return Players:GetCharacterAppearanceAsync(targetId) end)
        if successApp and appearance then
            for _, obj in ipairs(appearance:GetChildren()) do
                if obj:IsA("Accessory") or obj:IsA("Hat") or obj:IsA("Shirt") or obj:IsA("Pants") or obj:IsA("ShirtGraphic") or obj:IsA("CharacterMesh") or obj:IsA("BodyColors") then
                    local clone = obj:Clone()
                    clone.Parent = targetChar
                    table.insert(addedClothes, clone)
                elseif obj:IsA("Decal") and obj.Name == "face" then
                    local head = targetChar:FindFirstChild("Head")
                    if head then 
                        local clone = obj:Clone()
                        clone.Parent = head
                        table.insert(addedClothes, clone)
                    end
                elseif obj:IsA("SpecialMesh") or obj:IsA("BlockMesh") or obj:IsA("CylinderMesh") then
                    local head = targetChar:FindFirstChild("Head")
                    if head then 
                        local clone = obj:Clone()
                        clone.Parent = head 
                        table.insert(addedClothes, clone)
                    end
                end
            end
            appearance:Destroy()
        end

        modifiedPlayers[selectedPlayer] = {
            Type = "DirectR6",
            BackupClothes = backupClothes,
            AddedClothes = addedClothes
        }

        btn.Text = "Sucesso!"  
        task.delay(3, function() btn.Text = defaultText end)
        return -- FINALIZA AQUI! Não cria clone, não cria dummy, usa o corpo real!
    end

    -------------------------------------------------------------------------
    -- LÓGICA PARA R15 OU PARA R6 DENTRO DE UM MAPA R15 (USA DUMMY)
    -------------------------------------------------------------------------
    local rigType = (mode == "R15") and Enum.HumanoidRigType.R15 or Enum.HumanoidRigType.R6
    local modelSuccess, sourceModel = pcall(function()  
        return Players:CreateHumanoidModelFromUserId(targetId, rigType)  
    end)  

    if modelSuccess and sourceModel then  
        toggleOriginalCharacterVisibility(targetChar, false)  

        local dummy = sourceModel  
        dummy.Name = "VisualCloneDummy_" .. copyUsername  
          
        local fakeHum = dummy:FindFirstChildOfClass("Humanoid")  
        if fakeHum then  
            fakeHum.DisplayDistanceType = Enum.HumanoidDisplayDistanceType.None  
            fakeHum.PlatformStand = true  
        end  

        for _, part in ipairs(dummy:GetDescendants()) do  
            if part:IsA("BasePart") then  
                part.CanCollide = false
                part.Massless = true
                part.Anchored = false
                part.CanTouch = false
                part.CanQuery = false
            elseif part:IsA("Script") or part:IsA("LocalScript") then  
                part:Destroy()  
            end  
        end  

        dummy.Parent = targetChar  
        task.wait(0.1)  

        local realRoot = targetChar:FindFirstChild("HumanoidRootPart")  
        local fakeRoot = dummy:FindFirstChild("HumanoidRootPart")  
        local connection
        
        -- CLONE R15
        if mode == "R15" then
            if realRoot and fakeRoot then
                fakeRoot.CFrame = realRoot.CFrame
                local weld = Instance.new("WeldConstraint")
                weld.Part0 = realRoot
                weld.Part1 = fakeRoot
                weld.Parent = fakeRoot
            end

            connection = RunService.Stepped:Connect(function()  
                if not targetChar or not dummy or not dummy.Parent then return end  
                for _, realMotor in ipairs(targetChar:GetDescendants()) do  
                    if realMotor:IsA("Motor6D") then  
                        local fakeMotor = dummy:FindFirstChild(realMotor.Name, true)
                        if fakeMotor and fakeMotor:IsA("Motor6D") then  
                            fakeMotor.Transform = realMotor.Transform  
                        end
                    end  
                end  
            end)  
            task.spawn(function() applyTargetAnimations(targetChar, targetId) end)

        -- CLONE R6 (QUANDO O MAPA É R15 E NÃO DÁ PRA COLAR DIRETO)
        elseif mode == "R6" then
            if realRoot and fakeRoot then
                -- CORREÇÃO DA ALTURA (OFFSET +0.6) PARA NÃO AFUNDAR NA TERRA EM MAPAS R15!
                fakeRoot.CFrame = realRoot.CFrame * CFrame.new(0, 0.6, 0)
                local weld = Instance.new("WeldConstraint")
                weld.Part0 = realRoot
                weld.Part1 = fakeRoot
                weld.Parent = fakeRoot
            end

            local animator = fakeHum:FindFirstChildOfClass("Animator") or Instance.new("Animator", fakeHum)
            local r6AnimIDs = {
                Idle = "rbxassetid://180435571", Walk = "rbxassetid://180426354",
                Jump = "rbxassetid://125750702", Fall = "rbxassetid://180436148"
            }
            local loaded = {}
            for name, id in pairs(r6AnimIDs) do
                local a = Instance.new("Animation")
                a.AnimationId = id
                loaded[name] = animator:LoadAnimation(a)
            end
            
            local currentAnim = "Idle"
            loaded.Idle:Play()
            
            connection = RunService.RenderStepped:Connect(function()
                if not targetChar or not targetHum or not realRoot or not dummy or not dummy.Parent then return end
                local state = targetHum:GetState()
                local speed = (realRoot.Velocity * Vector3.new(1, 0, 1)).Magnitude
                local desiredAnim = "Idle"
                if state == Enum.HumanoidStateType.Freefall then desiredAnim = "Fall"
                elseif state == Enum.HumanoidStateType.Jumping then desiredAnim = "Jump"
                elseif speed > 1 then desiredAnim = "Walk" end
                
                if currentAnim ~= desiredAnim then
                    if loaded[currentAnim] then loaded[currentAnim]:Stop(0.2) end
                    if loaded[desiredAnim] then loaded[desiredAnim]:Play(0.2) end
                    currentAnim = desiredAnim
                end
                if currentAnim == "Walk" then loaded.Walk:AdjustSpeed(speed / 16) end
            end)
        end

        modifiedPlayers[selectedPlayer] = {  
            Type = "Clone",
            Dummy = dummy,  
            Connection = connection
        }  
        btn.Text = "Sucesso!"  
    else  
        btn.Text = "Erro ao Baixar!"  
    end  
    task.delay(3, function() btn.Text = defaultText end)
end
