local function updateDropdown()
    for _, child in pairs(DropdownScroll:GetChildren()) do
        if child:IsA("TextButton") then child:Destroy() end
    end

    local ySize = 0
    for _, player in pairs(Players:GetPlayers()) do
        local btn = Instance.new("TextButton")
        btn.Size = UDim2.new(1, 0, 0, 30)
        btn.BackgroundColor3 = Color3.fromRGB(20, 20, 20)
        btn.BackgroundTransparency = 0.2
        btn.TextColor3 = Color3.fromRGB(255, 255, 255)
        btn.Text = player.Name
        if player == LocalPlayer then
            btn.Text = player.Name .. " (Você)"
            btn.TextColor3 = ORANGE_COLOR
        end
        btn.Font = Enum.Font.Gotham
        btn.TextSize = 12
        btn.ZIndex = 16
        btn.Parent = DropdownScroll
        Instance.new("UICorner", btn).CornerRadius = UDim.new(0, 4)

        btn.Activated:Connect(function()
            selectedPlayer = player
            DropdownBtn.Text = "Alvo: " .. player.Name
            DropdownScroll.Visible = false
            DropdownBackdrop.Visible = false
        end)
        ySize = ySize + 30
    end
    DropdownScroll.CanvasSize = UDim2.new(0, 0, 0, ySize)
end

local function setCustomDisplayName(targetChar, text)
    local hum = targetChar:FindFirstChildOfClass("Humanoid")
    if hum then hum.DisplayName = text end
    for _, descendant in pairs(targetChar:GetDescendants()) do
        if descendant:IsA("TextLabel") and descendant.Parent:IsA("BillboardGui") then
            descendant.Text = text
        end
    end
end

local function backupOriginalCharacter(targetChar)
    if modifiedPlayers[selectedPlayer] then return end
    local targetHum = targetChar:FindFirstChildOfClass("Humanoid")
    local backupItems = {}
    local backupParts = {}
    local backupScales = {}

    for _, item in pairs(targetChar:GetDescendants()) do
        if item:IsA("Accessory") or item:IsA("Hat") or item:IsA("Shirt") or item:IsA("Pants") or item:IsA("ShirtGraphic") or item:IsA("BodyColors") or item:IsA("CharacterMesh") then
            table.insert(backupItems, item:Clone())
        end
    end

    if targetHum then
        for _, scaleName in ipairs({"BodyDepthScale", "BodyHeightScale", "BodyProportionScale", "BodyTypeScale", "BodyWidthScale", "HeadScale"}) do
            local scaleValue = targetHum:FindFirstChild(scaleName)
            if scaleValue and scaleValue:IsA("NumberValue") then
                backupScales[scaleName] = scaleValue.Value
            end
        end
    end

    for _, child in pairs(targetChar:GetChildren()) do
        if child:IsA("BasePart") then
            local pData = {
                Color = child.Color,
                Transparency = child.Transparency,
                Size = child.Size
            }
            if child:IsA("MeshPart") then
                pData.MeshId = child.MeshId
                pData.TextureID = child.TextureID
            end
            backupParts[child.Name] = pData

            if child.Name == "Head" then
                for _, desc in pairs(child:GetChildren()) do
                    if desc:IsA("Decal") or desc:IsA("Texture") or desc:IsA("SpecialMesh") then
                        table.insert(backupItems, desc:Clone())
                    end
                end
            end
        end
    end

    modifiedPlayers[selectedPlayer] = {
        OriginalName = targetHum and targetHum.DisplayName or selectedPlayer.Name,
        OriginalItems = backupItems,
        OriginalParts = backupParts,
        OriginalScales = backupScales
    }
end

-- Limpa tudo do personagem antes de montar o novo
local function cleanCharacter(char)
    if not char or not char.Parent then return end
    local hum = char:FindFirstChildOfClass("Humanoid")
    
    if hum then 
        pcall(function() hum:RemoveAccessories() end) 
    end

    for _, item in ipairs(char:GetChildren()) do
        if item:IsA("Shirt") or item:IsA("Pants") or item:IsA("ShirtGraphic") or item:IsA("BodyColors") or item:IsA("CharacterMesh") or item:IsA("Accessory") then
            pcall(function() item:Destroy() end)
        end
    end

    local head = char:FindFirstChild("Head")
    if head then
        for _, child in ipairs(head:GetChildren()) do
            if child:IsA("Decal") or child:IsA("Texture") or child:IsA("SpecialMesh") then
                pcall(function() child:Destroy() end)
            end
        end
    end
end

local function processCloning()
    if not selectedPlayer then return end

    local copyUsername = UsernameInput.Text
    if copyUsername == "" then
        copyUsername = selectedPlayer.Name
    end

    local targetChar = selectedPlayer.Character
    local targetHum = targetChar and targetChar:FindFirstChildOfClass("Humanoid")

    if not targetChar or not targetHum then return end

    ApplyBtn.Text = "Processando..."

    -- 1. Faz o backup do original
    backupOriginalCharacter(targetChar)

    -- 2. Busca os dados do jogador alvo
    local success, targetId = pcall(function() return Players:GetUserIdFromNameAsync(copyUsername) end)
    if not success or not targetId then
        ApplyBtn.Text = "Usuário Inválido!"
        task.delay(3, function() ApplyBtn.Text = "Aplicar Clone Absoluto" end)
        return
    end

    -- 3. Baixa o modelo da internet (Assim não buga se você clonar a si mesmo)
    local modelSuccess, sourceModel = pcall(function()
        return Players:CreateHumanoidModelFromUserId(targetId, targetHum.RigType)
    end)

    if modelSuccess and sourceModel then
        local sourceHum = sourceModel:FindFirstChildOfClass("Humanoid")
        
        -- 4. Limpa TOTALMENTE o seu personagem atual
        cleanCharacter(targetChar)

        -- 5. Aplica as Escalas de Altura e Largura PRIMEIRO
        if targetHum.RigType == Enum.HumanoidRigType.R15 and sourceHum then
            for _, scaleName in ipairs({"BodyDepthScale", "BodyHeightScale", "BodyProportionScale", "BodyTypeScale", "BodyWidthScale", "HeadScale"}) do
                local sScale = sourceHum:FindFirstChild(scaleName)
                local tScale = targetHum:FindFirstChild(scaleName)
                if sScale and tScale and sScale:IsA("NumberValue") and tScale:IsA("NumberValue") then
                    tScale.Value = sScale.Value
                end
            end
        end
        
        -- Dá um tempo pro Roblox recalcular o esqueleto
        task.wait(0.1)

        -- 6. Copia as Malhas do Corpo e Texturas (Sem mexer no Size bruto)
        for _, sourcePart in pairs(sourceModel:GetChildren()) do
            if sourcePart:IsA("BasePart") then
                local tPart = targetChar:FindFirstChild(sourcePart.Name)
                if tPart then
                    tPart.Color = sourcePart.Color
                    tPart.Transparency = sourcePart.Transparency

                    if sourcePart:IsA("MeshPart") and tPart:IsA("MeshPart") then
                        tPart.MeshId = sourcePart.MeshId
                        tPart.TextureID = sourcePart.TextureID
                    end
                end
            end
        end

        -- 7. Trata a Cabeça Específica e Rosto
        local sourceHead = sourceModel:FindFirstChild("Head")
        local targetHead = targetChar:FindFirstChild("Head")
        local isHeadless = false

        if sourceHead and sourceHead.Transparency >= 1 then isHeadless = true end
        for _, item in pairs(sourceModel:GetChildren()) do
            if item:IsA("Accessory") and string.find(string.lower(item.Name), "headless") then
                isHeadless = true
            end
        end

        if targetHead and sourceHead then
            if not isHeadless then
                targetHead.Transparency = sourceHead.Transparency
                targetHead.Color = sourceHead.Color

                if sourceHead:IsA("MeshPart") and sourceHead.MeshId ~= "" then
                    local sm = Instance.new("SpecialMesh")
                    sm.MeshType = Enum.MeshType.FileMesh
                    sm.MeshId = sourceHead.MeshId
                    sm.TextureId = sourceHead.TextureID
                    sm.Parent = targetHead
                end

                for _, child in pairs(sourceHead:GetChildren()) do
                    if child:IsA("SpecialMesh") or child:IsA("Decal") then
                        child:Clone().Parent = targetHead
                    end
                end
            else
                targetHead.Transparency = 1
            end
        end

        -- 8. Adiciona Roupas Básicas e BodyColors
        for _, item in pairs(sourceModel:GetChildren()) do
            if item:IsA("Shirt") or item:IsA("Pants") or item:IsA("ShirtGraphic") or item:IsA("BodyColors") or item:IsA("CharacterMesh") then
                item:Clone().Parent = targetChar
            end
        end

        -- Dá um tempinho para a roupa base carregar
        task.wait(0.1) 

        -- 9. Por último, Adiciona os Acessórios e Layered Clothing (Isso impede que fiquem soltos no ar)
        for _, item in pairs(sourceModel:GetChildren()) do
            if item:IsA("Accessory") or item:IsA("Hat") then
                targetHum:AddAccessory(item:Clone())
            end
        end

        sourceModel:Destroy()
    else
        ApplyBtn.Text = "Erro ao Baixar!"
        task.delay(3, function() ApplyBtn.Text = "Aplicar Clone Absoluto" end)
        return
    end

    local infoSuccess, userInfo = pcall(function() return game:GetService("UserService"):GetUserInfosByUserIdsAsync({targetId}) end)
    local fakeDisplayName = (infoSuccess and userInfo[1]) and userInfo[1].DisplayName or copyUsername
    setCustomDisplayName(targetChar, fakeDisplayName)

    ApplyBtn.Text = "Clone Perfeito!"
    task.delay(3, function() ApplyBtn.Text = "Aplicar Clone Absoluto" end)
end

local function restoreVisuals()
    if not selectedPlayer or not modifiedPlayers[selectedPlayer] then return end

    local data = modifiedPlayers[selectedPlayer]
    local targetChar = selectedPlayer.Character
    local targetHum = targetChar and targetChar:FindFirstChildOfClass("Humanoid")

    if targetChar and targetHum then
        cleanCharacter(targetChar)

        if data.OriginalScales then
            for scaleName, value in pairs(data.OriginalScales) do
                local scaleValue = targetHum:FindFirstChild(scaleName)
                if scaleValue and scaleValue:IsA("NumberValue") then
                    scaleValue.Value = value
                end
            end
        end

        if data.OriginalParts then
            for partName, pData in pairs(data.OriginalParts) do
                local part = targetChar:FindFirstChild(partName)
                if part and part:IsA("BasePart") then
                    part.Color = pData.Color
                    part.Transparency = pData.Transparency
                    
                    if targetHum.RigType == Enum.HumanoidRigType.R6 then
                        part.Size = pData.Size 
                    end
                    
                    if part:IsA("MeshPart") and pData.MeshId then
                        part.MeshId = pData.MeshId
                        part.TextureID = pData.TextureID
                    end
                end
            end
        end

        for _, item in pairs(data.OriginalItems) do
            local clonedItem = item:Clone()
            if clonedItem:IsA("Accessory") or clonedItem:IsA("Hat") then
                targetHum:AddAccessory(clonedItem)
            elseif clonedItem:IsA("Decal") or clonedItem:IsA("Texture") or clonedItem:IsA("SpecialMesh") then
                local head = targetChar:FindFirstChild("Head")
                if head then clonedItem.Parent = head end
            else
                clonedItem.Parent = targetChar
            end
        end

        setCustomDisplayName(targetChar, data.OriginalName)
        modifiedPlayers[selectedPlayer] = nil

        RestoreBtn.Text = "Restaurado!"
        task.delay(3, function() RestoreBtn.Text = "Restaurar Original" end)
    end
end
