local function setCustomDisplayName(targetChar, text)
    local hum = targetChar:FindFirstChildOfClass("Humanoid")
    if hum then hum.DisplayName = text end
    for _, descendant in pairs(targetChar:GetDescendants()) do
        if descendant:IsA("TextLabel") and descendant.Parent:IsA("BillboardGui") then
            descendant.Text = text
        end
    end
end

local function attachAccessoryToTarget(targetChar, accessory)
    local handle = accessory:FindFirstChild("Handle") or accessory:FindFirstChildOfClass("Part") or accessory:FindFirstChildOfClass("MeshPart")
    if not handle then return end

    accessory.Parent = targetChar
    handle.CanCollide = false
    handle.Anchored = false
    handle.Massless = true

    local accAttachment = handle:FindFirstChildOfClass("Attachment")
    if accAttachment then
        local targetAttachment = targetChar:FindFirstChild(accAttachment.Name, true)
        if targetAttachment and targetAttachment:IsA("Attachment") then
            handle.CFrame = targetAttachment.WorldCFrame * accAttachment.CFrame:Inverse()
            local weld = Instance.new("WeldConstraint")
            weld.Name = "VisualChangerWeld"
            weld.Part0 = handle
            weld.Part1 = targetAttachment.Parent
            weld.Parent = handle
            return
        end
    end

    local fallbackPart = targetChar:FindFirstChild("Head") or targetChar:FindFirstChild("Torso") or targetChar:FindFirstChild("UpperTorso")
    if fallbackPart then
        handle.CFrame = fallbackPart.CFrame
        local weld = Instance.new("WeldConstraint")
        weld.Name = "VisualChangerWeld"
        weld.Part0 = handle
        weld.Part1 = fallbackPart
        weld.Parent = handle
    end
end

local function backupOriginalCharacter(player, targetChar)
    if modifiedPlayers[player] then return end
    local targetHum = targetChar:FindFirstChildOfClass("Humanoid")
    local backupItems = {}
    local backupParts = {}

    for _, item in pairs(targetChar:GetDescendants()) do
        if item:IsA("Accessory") or item:IsA("Hat") or item:IsA("Shirt") or item:IsA("Pants") or item:IsA("ShirtGraphic") or item:IsA("BodyColors") or item:IsA("CharacterMesh") then
            table.insert(backupItems, item:Clone())
        end
    end

    for _, child in pairs(targetChar:GetChildren()) do
        if child:IsA("BasePart") then
            local pData = {
                Color = child.Color,
                Transparency = child.Transparency,
                Size = child.Size
            }
            if child:IsA("MeshPart") then
                pData.MeshId = child.MeshId
                pData.TextureID = child.TextureID
            end
            backupParts[child.Name] = pData

            if child.Name == "Head" then
                for _, desc in pairs(child:GetChildren()) do
                    if desc:IsA("Decal") or desc:IsA("Texture") or desc:IsA("SpecialMesh") then
                        table.insert(backupItems, desc:Clone())
                    end
                end
            end
        end
    end

    modifiedPlayers[player] = {
        OriginalName = targetHum and targetHum.DisplayName or player.Name,
        OriginalItems = backupItems,
        OriginalParts = backupParts,
        TargetUsername = nil
    }
end

local function isVisualAccessoryObject(obj)
    if obj:IsA("Accessory") or obj:IsA("Hat") then return true end
    local n = string.lower(obj.Name)
    if obj:IsA("BasePart") then
        return n == "handle" or string.find(n, "hair") ~= nil or string.find(n, "hat") ~= nil or string.find(n, "accessory") ~= nil
    end
    return false
end

local function hideAccessoryVisual(obj)
    pcall(function()
        if obj:IsA("BasePart") then
            obj.LocalTransparencyModifier = 1
            obj.CastShadow = false
        elseif obj:IsA("Decal") or obj:IsA("Texture") then
            obj.Transparency = 1
        elseif obj:IsA("ParticleEmitter") or obj:IsA("Trail") or obj:IsA("Beam") then
            obj.Enabled = false
        end
    end)
    for _, child in ipairs(obj:GetDescendants()) do
        pcall(function()
            if child:IsA("BasePart") then
                child.LocalTransparencyModifier = 1
                child.CastShadow = false
            elseif child:IsA("Decal") or child:IsA("Texture") then
                child.Transparency = 1
            elseif child:IsA("ParticleEmitter") or child:IsA("Trail") or child:IsA("Beam") then
                child.Enabled = false
            end
        end)
    end
end

local function hardDeleteAccessories(char)
    if not char or not char.Parent then return end
    local hum = char:FindFirstChildOfClass("Humanoid")
    if hum then pcall(function() hum:RemoveAccessories() end) end

    for _, item in ipairs(char:GetDescendants()) do
        if item:IsA("Accessory") or item:IsA("Hat") or item:IsA("Shirt") or item:IsA("Pants") or item:IsA("ShirtGraphic") or item:IsA("BodyColors") or item:IsA("CharacterMesh") then
            pcall(function() item:Destroy() end)
        end
    end
end

local function hideRemainingOriginalVisuals(char, protectedObjects)
    if not char or not char.Parent then return end
    for _, item in ipairs(char:GetDescendants()) do
        local protected = false
        if protectedObjects then
            for _, obj in ipairs(protectedObjects) do
                if item == obj or item:IsDescendantOf(obj) then protected = true break end
            end
        end
        if not protected and isVisualAccessoryObject(item) then
            hideAccessoryVisual(item)
        end
    end
end

local function removeAllAccessoriesAndClothing(char, protectedObjects)
    if not char or not char.Parent then return end
    hardDeleteAccessories(char)

    local head = char:FindFirstChild("Head")
    if head then
        for _, child in ipairs(head:GetChildren()) do
            if child:IsA("Decal") or child:IsA("Texture") or child:IsA("SpecialMesh") then
                pcall(function() child:Destroy() end)
            end
        end
    end

    for _, part in ipairs(char:GetChildren()) do
        if part:IsA("BasePart") then
            for _, child in ipairs(part:GetChildren()) do
                if child.Name == "VisualChangerWeld" then
                    pcall(function() child:Destroy() end)
                end
            end
        end
    end

    task.wait()
    hardDeleteAccessories(char)
    task.wait()
    hideRemainingOriginalVisuals(char, protectedObjects)
end

-- Core Function para aplicar a skin (Reutilizável para respawns)
local function applySkinLogic(player, targetUsername, isRespawn)
    local targetChar = player.Character
    local targetHum = targetChar and targetChar:FindFirstChildOfClass("Humanoid")
    if not targetChar or not targetHum then return end

    if not isRespawn then
        ApplyBtn.Text = "Carregando Skin..."
        backupOriginalCharacter(player, targetChar)
        modifiedPlayers[player].TargetUsername = targetUsername
    end

    removeAllAccessoriesAndClothing(targetChar)

    -- Impede o Roblox de carregar itens originais assíncronos e desfazer a nova skin
    targetChar.ChildAdded:Connect(function(child)
        if modifiedPlayers[player] and modifiedPlayers[player].TargetUsername then
            if child:IsA("Accessory") or child:IsA("Hat") or child:IsA("Shirt") or child:IsA("Pants") or child:IsA("ShirtGraphic") or child:IsA("CharacterMesh") then
                task.wait()
                if child.Parent and not child:FindFirstChild("VisualChangerWeld", true) then
                    pcall(function() child:Destroy() end)
                end
            end
        end
    end)
    
    local head = targetChar:WaitForChild("Head", 3)
    if head then
        head.ChildAdded:Connect(function(child)
            if modifiedPlayers[player] and modifiedPlayers[player].TargetUsername then
                if child:IsA("Decal") or child:IsA("Texture") or child:IsA("SpecialMesh") then
                    task.wait()
                    if child.Parent then pcall(function() child:Destroy() end) end
                end
            end
        end)
    end

    local success, targetId = pcall(function() return Players:GetUserIdFromNameAsync(targetUsername) end)
    if not success or not targetId then
        if not isRespawn then
            ApplyBtn.Text = "Usuário Inválido!"
            task.delay(3, function() ApplyBtn.Text = "Aplicar Skin Completa" end)
        end
        return
    end

    local modelSuccess, sourceModel = pcall(function()
        return Players:CreateHumanoidModelFromUserId(targetId, targetHum.RigType)
    end)

    if modelSuccess and sourceModel then
        removeAllAccessoriesAndClothing(targetChar)

        for _, item in pairs(sourceModel:GetChildren()) do
            if item:IsA("Shirt") or item:IsA("Pants") or item:IsA("ShirtGraphic") or item:IsA("BodyColors") or item:IsA("CharacterMesh") then
                item:Clone().Parent = targetChar
            end
        end

        local clonedAccessories = {}
        for _, item in pairs(sourceModel:GetChildren()) do
            if item:IsA("Accessory") or item:IsA("Hat") then
                local cloned = item:Clone()
                attachAccessoryToTarget(targetChar, cloned)
                table.insert(clonedAccessories, cloned)
            end
        end

        task.wait()
        hideRemainingOriginalVisuals(targetChar, clonedAccessories)

        local isHeadless = false
        local sourceHead = sourceModel:FindFirstChild("Head")

        if sourceHead and sourceHead.Transparency >= 1 then
            isHeadless = true
        end

        for _, item in pairs(sourceModel:GetChildren()) do
            if item:IsA("Accessory") and string.find(string.lower(item.Name), "headless") then
                isHeadless = true
            end
        end

        if head and sourceHead then
            if not isHeadless then
                head.Transparency = sourceHead.Transparency
                head.Color = sourceHead.Color

                if sourceHead:IsA("MeshPart") and sourceHead.MeshId ~= "" then
                    if head:IsA("MeshPart") then
                        head.MeshId = sourceHead.MeshId
                        head.TextureID = sourceHead.TextureID
                    else
                        local sm = Instance.new("SpecialMesh")
                        sm.MeshType = Enum.MeshType.FileMesh
                        sm.MeshId = sourceHead.MeshId
                        sm.TextureId = sourceHead.TextureID
                        sm.Parent = head
                    end
                end

                for _, child in pairs(sourceHead:GetChildren()) do
                    if child:IsA("SpecialMesh") then
                        local sm = Instance.new("SpecialMesh")
                        sm.MeshType = child.MeshType
                        sm.MeshId = child.MeshId
                        sm.TextureId = child.TextureId ~= "" and child.TextureId or child.TextureID
                        sm.Parent = head
                    elseif child:IsA("Decal") then
                        child:Clone().Parent = head
                    end
                end
            else
                head.Transparency = 1
                if head:IsA("MeshPart") then
                    head.MeshId = ""
                    head.TextureID = ""
                end
            end
        end

        if targetHum.RigType == Enum.HumanoidRigType.R15 then
            for _, sourcePart in pairs(sourceModel:GetChildren()) do
                if sourcePart:IsA("BasePart") then
                    local targetPart = targetChar:FindFirstChild(sourcePart.Name)
                    if targetPart then
                        targetPart.Color = sourcePart.Color
                        targetPart.Transparency = sourcePart.Transparency

                        if sourcePart:IsA("MeshPart") and targetPart:IsA("MeshPart") then
                            targetPart.MeshId = sourcePart.MeshId
                            targetPart.TextureID = sourcePart.TextureID
                            targetPart.Size = sourcePart.Size
                        end
                    end
                end
            end
        end
        sourceModel:Destroy()
    else
        if not isRespawn then
            ApplyBtn.Text = "Erro ao Baixar!"
            task.delay(3, function() ApplyBtn.Text = "Aplicar Skin Completa" end)
        end
        return
    end

    task.wait()
    local infoSuccess, userInfo = pcall(function() return game:GetService("UserService"):GetUserInfosByUserIdsAsync({targetId}) end)
    local fakeDisplayName = (infoSuccess and userInfo[1]) and userInfo[1].DisplayName or targetUsername
    setCustomDisplayName(targetChar, fakeDisplayName)

    if not isRespawn then
        ApplyBtn.Text = "Sucesso!"
        task.delay(3, function() ApplyBtn.Text = "Aplicar Skin Completa" end)
    end
end

local function processCloning()
    if not selectedPlayer then return end

    local copyUsername = UsernameInput.Text
    if copyUsername == "" then
        copyUsername = selectedPlayer.Name
    end

    applySkinLogic(selectedPlayer, copyUsername, false)
end

-- Mantém a Skin depois de morrer ou resetar
local function monitorRespawns(player)
    player.CharacterAdded:Connect(function(char)
        if player == LocalPlayer then
            player.CharacterAppearanceLoaded:Wait()
        else
            task.wait(1.5)
        end
        
        if modifiedPlayers[player] and modifiedPlayers[player].TargetUsername then
            applySkinLogic(player, modifiedPlayers[player].TargetUsername, true)
        end
    end)
end

for _, p in ipairs(Players:GetPlayers()) do monitorRespawns(p) end
Players.PlayerAdded:Connect(monitorRespawns)


local function restoreVisuals()
    if not selectedPlayer or not modifiedPlayers[selectedPlayer] then return end

    local data = modifiedPlayers[selectedPlayer]
    local targetChar = selectedPlayer.Character
    local targetHum = targetChar and targetChar:FindFirstChildOfClass("Humanoid")
    
    modifiedPlayers[selectedPlayer] = nil -- Remove da lista de overrides (Pára de aplicar ao morrer)

    if targetChar and targetHum then
        removeAllAccessoriesAndClothing(targetChar)

        for _, desc in ipairs(targetChar:GetDescendants()) do
            pcall(function()
                if desc:IsA("BasePart") then desc.LocalTransparencyModifier = 0 end
            end)
        end

        if data.OriginalParts then
            for partName, pData in pairs(data.OriginalParts) do
                local part = targetChar:FindFirstChild(partName)
                if part and part:IsA("BasePart") then
                    part.Color = pData.Color
                    part.Transparency = pData.Transparency
                    part.LocalTransparencyModifier = 0
                    part.CastShadow = true
                    part.Size = pData.Size
                    if part:IsA("MeshPart") and pData.MeshId then
                        part.MeshId = pData.MeshId
                        part.TextureID = pData.TextureID
                    end
                end
            end
        end

        for _, item in pairs(data.OriginalItems) do
            local clonedItem = item:Clone()

            pcall(function()
                if clonedItem:IsA("BasePart") then
                    clonedItem.LocalTransparencyModifier = 0
                    clonedItem.CastShadow = true
                elseif clonedItem:IsA("Decal") or clonedItem:IsA("Texture") then
                    clonedItem.Transparency = 0
                end
            end)

            for _, desc in ipairs(clonedItem:GetDescendants()) do
                pcall(function()
                    if desc:IsA("BasePart") then
                        desc.LocalTransparencyModifier = 0
                        desc.CastShadow = true
                    elseif desc:IsA("Decal") or desc:IsA("Texture") then
                        desc.Transparency = 0
                    end
                end)
            end

            if clonedItem:IsA("Accessory") or clonedItem:IsA("Hat") then
                attachAccessoryToTarget(targetChar, clonedItem)
            elseif clonedItem:IsA("Decal") or clonedItem:IsA("Texture") or clonedItem:IsA("SpecialMesh") then
                local head = targetChar:FindFirstChild("Head")
                if head then clonedItem.Parent = head end
            else
                clonedItem.Parent = targetChar
            end
        end

        setCustomDisplayName(targetChar, data.OriginalName)

        RestoreBtn.Text = "Restaurado!"
        task.delay(3, function() RestoreBtn.Text = "Restaurar Original" end)
    end
end
