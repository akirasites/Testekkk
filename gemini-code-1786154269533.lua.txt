local function updateDropdown()
    for _, child in pairs(DropdownScroll:GetChildren()) do
        if child:IsA("TextButton") then child:Destroy() end
    end
    
    local ySize = 0
    for _, player in pairs(Players:GetPlayers()) do
        if player ~= LocalPlayer then
            local btn = Instance.new("TextButton")
            btn.Size = UDim2.new(1, 0, 0, 30)
            btn.BackgroundColor3 = Color3.fromRGB(20, 20, 20)
            btn.BackgroundTransparency = 0.2
            btn.TextColor3 = Color3.fromRGB(255, 255, 255)
            btn.Text = player.Name
            btn.Font = Enum.Font.Gotham
            btn.TextSize = 12
            btn.ZIndex = 16
            btn.Parent = DropdownScroll
            
            local btnCorner = Instance.new("UICorner")
            btnCorner.CornerRadius = UDim.new(0, 4)
            btnCorner.Parent = btn
            
            btn.Activated:Connect(function()
                selectedPlayer = player
                DropdownBtn.Text = "Alvo: " .. player.Name
                DropdownScroll.Visible = false
                DropdownBackdrop.Visible = false
            end)
            ySize = ySize + 30
        end
    end
    DropdownScroll.CanvasSize = UDim2.new(0, 0, 0, ySize)
end

Players.PlayerAdded:Connect(updateDropdown)

local function setCustomDisplayName(targetChar, text)
    local hum = targetChar:FindFirstChild("Humanoid")
    if hum then hum.DisplayName = text end
    for _, descendant in pairs(targetChar:GetDescendants()) do
        if descendant:IsA("TextLabel") and descendant.Parent:IsA("BillboardGui") then
            descendant.Text = text
        end
    end
end

local function attachAccessoryToTarget(targetChar, accessory)
    local handle = accessory:FindFirstChild("Handle") or accessory:FindFirstChildOfClass("Part") or accessory:FindFirstChildOfClass("MeshPart")
    if not handle then return end
    
    accessory.Parent = targetChar
    handle.CanCollide = false
    handle.Anchored = false
    handle.Massless = true
    
    local accAttachment = handle:FindFirstChildOfClass("Attachment")
    if accAttachment then
        local targetAttachment = targetChar:FindFirstChild(accAttachment.Name, true)
        if targetAttachment and targetAttachment:IsA("Attachment") then
            handle.CFrame = targetAttachment.WorldCFrame * accAttachment.CFrame:Inverse()
            local weld = Instance.new("WeldConstraint")
            weld.Name = "VisualChangerWeld"
            weld.Part0 = handle
            weld.Part1 = targetAttachment.Parent
            weld.Parent = handle
            return
        end
    end
    
    local fallbackPart = targetChar:FindFirstChild("Head") or targetChar:FindFirstChild("Torso") or targetChar:FindFirstChild("UpperTorso")
    if fallbackPart then
        handle.CFrame = fallbackPart.CFrame
        local weld = Instance.new("WeldConstraint")
        weld.Name = "VisualChangerWeld"
        weld.Part0 = handle
        weld.Part1 = fallbackPart
        weld.Parent = handle
    end
end

local function processCloning()
    if not selectedPlayer then return end

    local copyUsername = UsernameInput.Text
    if copyUsername == "" then
        copyUsername = selectedPlayer.Name
    end
    
    local targetChar = selectedPlayer.Character
    local targetHum = targetChar and targetChar:FindFirstChild("Humanoid")
    
    if not targetChar or not targetHum then return end

    local originalText = ApplyBtn.Text
    ApplyBtn.Text = "Carregando Skin..."

    if not modifiedPlayers[selectedPlayer] then
        local backupItems = {}
        for _, item in pairs(targetChar:GetDescendants()) do
            if item:IsA("Accessory") or item:IsA("Hat") or item:IsA("Accoutrement") or item:IsA("Shirt") or item:IsA("Pants") or item:IsA("ShirtGraphic") or item:IsA("BodyColors") or item:IsA("CharacterMesh") or item:IsA("WrapLayer") or item:IsA("Clothing") then
                table.insert(backupItems, item:Clone())
            end
        end
        
        local targetHead = targetChar:FindFirstChild("Head")
        local headBackup = {
            Transparency = targetHead and targetHead.Transparency or 0,
            Color = targetHead and targetHead.Color or Color3.new(1,1,1),
            Children = {}
        }
        if targetHead then
            if targetHead:IsA("MeshPart") then
                headBackup.MeshId = targetHead.MeshId
                headBackup.TextureID = targetHead.TextureID
                headBackup.Size = targetHead.Size
            end
            for _, child in pairs(targetHead:GetChildren()) do
                table.insert(headBackup.Children, child:Clone())
            end
        end
        
        modifiedPlayers[selectedPlayer] = {
            OriginalName = targetHum.DisplayName,
            OriginalItems = backupItems,
            HeadBackup = headBackup
        }
    end

    for _, item in pairs(targetChar:GetDescendants()) do
        if item:IsA("Accessory") or item:IsA("Hat") or item:IsA("Accoutrement") or item:IsA("Shirt") or item:IsA("Pants") or item:IsA("ShirtGraphic") or item:IsA("BodyColors") or item:IsA("CharacterMesh") or item:IsA("WrapLayer") or item:IsA("Clothing") then
            item:Destroy()
        end
    end
    
    local targetHead = targetChar:FindFirstChild("Head")
    if targetHead then
        for _, child in pairs(targetHead:GetChildren()) do
            child:Destroy()
        end
    end
    
    for _, part in pairs(targetChar:GetChildren()) do
        if part:IsA("BasePart") then
            for _, child in pairs(part:GetChildren()) do
                if child.Name == "VisualChangerWeld" then
                    child:Destroy()
                end
            end
        end
    end

    local success, targetId = pcall(function() return Players:GetUserIdFromNameAsync(copyUsername) end)
    if not success or not targetId then
        ApplyBtn.Text = "Usuário Inválido!"
        task.wait(1.5)
        ApplyBtn.Text = originalText
        return
    end

    local modelSuccess, sourceModel = pcall(function()
        return Players:CreateHumanoidModelFromUserId(targetId)
    end)

    if modelSuccess and sourceModel then
        for _, item in pairs(targetChar:GetDescendants()) do
            if item:IsA("Accessory") or item:IsA("Hat") or item:IsA("Accoutrement") or item:IsA("Shirt") or item:IsA("Pants") or item:IsA("ShirtGraphic") or item:IsA("BodyColors") or item:IsA("CharacterMesh") or item:IsA("WrapLayer") or item:IsA("Clothing") then
                item:Destroy()
            end
        end

        for _, item in pairs(sourceModel:GetChildren()) do
            if item:IsA("Shirt") or item:IsA("Pants") or item:IsA("ShirtGraphic") or item:IsA("BodyColors") or item:IsA("CharacterMesh") or item:IsA("WrapLayer") then
                item:Clone().Parent = targetChar
            end
        end

        for _, item in pairs(sourceModel:GetChildren()) do
            if item:IsA("Accessory") or item:IsA("Hat") or item:IsA("Accoutrement") then
                attachAccessoryToTarget(targetChar, item:Clone())
            end
        end

        local isHeadless = false
        local sourceHead = sourceModel:FindFirstChild("Head")
        
        if sourceHead and sourceHead.Transparency >= 1 then
            isHeadless = true
        end
        
        for _, item in pairs(sourceModel:GetChildren()) do
            if (item:IsA("Accessory") or item:IsA("Hat") or item:IsA("Accoutrement")) and string.find(string.lower(item.Name), "headless") then
                isHeadless = true
            end
        end

        if targetHead and sourceHead then
            if not isHeadless then
                targetHead.Transparency = sourceHead.Transparency
                targetHead.Color = sourceHead.Color
                
                if targetHead:IsA("MeshPart") then
                    if sourceHead:IsA("MeshPart") and sourceHead.MeshId ~= "" then
                        targetHead.MeshId = sourceHead.MeshId
                        targetHead.TextureID = sourceHead.TextureID
                        targetHead.Size = sourceHead.Size
                    else
                        targetHead.MeshId = ""
                        targetHead.TextureID = ""
                    end
                end

                if sourceHead:IsA("MeshPart") and sourceHead.MeshId ~= "" then
                    local sm = targetHead:FindFirstChildOfClass("SpecialMesh")
                    if not sm then
                        sm = Instance.new("SpecialMesh")
                        sm.Parent = targetHead
                    end
                    sm.MeshType = Enum.MeshType.FileMesh
                    sm.MeshId = sourceHead.MeshId
                    sm.TextureId = sourceHead.TextureID
                else
                    local sm = targetHead:FindFirstChildOfClass("SpecialMesh")
                    if sm then sm:Destroy() end
                end

                for _, child in pairs(sourceHead:GetChildren()) do
                    if child:IsA("SpecialMesh") then
                        local sm = targetHead:FindFirstChildOfClass("SpecialMesh")
                        if not sm then
                            sm = Instance.new("SpecialMesh")
                            sm.Parent = targetHead
                        end
                        sm.MeshType = child.MeshType
                        sm.MeshId = child.MeshId
                        sm.TextureId = child.TextureId ~= "" and child.TextureId or child.TextureID
                    elseif child:IsA("Decal") then
                        child:Clone().Parent = targetHead
                    end
                end
            else
                targetHead.Transparency = 1
                local sm = targetHead:FindFirstChildOfClass("SpecialMesh")
                if sm then sm:Destroy() end
            end
        end

        for _, sourcePart in pairs(sourceModel:GetChildren()) do
            if sourcePart:IsA("BasePart") then
                local targetPart = targetChar:FindFirstChild(sourcePart.Name)
                if targetPart then
                    targetPart.Color = sourcePart.Color
                    targetPart.Transparency = sourcePart.Transparency
                    
                    if targetPart:IsA("MeshPart") then
                        if sourcePart:IsA("MeshPart") and sourcePart.MeshId ~= "" then
                            targetPart.MeshId = sourcePart.MeshId
                            targetPart.TextureID = sourcePart.TextureID
                            targetPart.Size = sourcePart.Size
                        else
                            targetPart.MeshId = ""
                            targetPart.TextureID = ""
                        end
                    end
                end
            end
        end

        local r6LeftLeg = targetChar:FindFirstChild("Left Leg")
        local r15LeftLegPart = sourceModel:FindFirstChild("LeftLowerLeg") or sourceModel:FindFirstChild("LeftFoot")
        if r6LeftLeg and r15LeftLegPart and r15LeftLegPart:IsA("MeshPart") then
            for _, child in pairs(r6LeftLeg:GetChildren()) do
                if child:IsA("SpecialMesh") then child:Destroy() end
            end
            local sm = Instance.new("SpecialMesh")
            sm.MeshType = Enum.MeshType.FileMesh
            sm.MeshId = r15LeftLegPart.MeshId
            sm.TextureID = r15LeftLegPart.TextureID
            sm.Parent = r6LeftLeg
            r6LeftLeg.Color = r15LeftLegPart.Color
            r6LeftLeg.Transparency = r15LeftLegPart.Transparency
        elseif r6LeftLeg then
            for _, child in pairs(r6LeftLeg:GetChildren()) do
                if child:IsA("SpecialMesh") then child:Destroy() end
            end
        end

        sourceModel:Destroy()
    else
        ApplyBtn.Text = "Erro ao Baixar!"
        task.wait(1.5)
        ApplyBtn.Text = originalText
        return
    end

    local infoSuccess, userInfo = pcall(function() return game:GetService("UserService"):GetUserInfosByUserIdsAsync({targetId}) end)
    local fakeDisplayName = (infoSuccess and userInfo[1]) and userInfo[1].DisplayName or copyUsername
    setCustomDisplayName(targetChar, fakeDisplayName)

    ApplyBtn.Text = "Sucesso!"
    task.wait(1.5)
    ApplyBtn.Text = originalText
end

local function restoreVisuals()
    if not selectedPlayer or not modifiedPlayers[selectedPlayer] then return end
    
    local data = modifiedPlayers[selectedPlayer]
    local targetChar = selectedPlayer.Character
    local targetHum = targetChar and targetChar:FindFirstChild("Humanoid")
    
    if targetChar and targetHum then
        for _, item in pairs(targetChar:GetDescendants()) do
            if item:IsA("Accessory") or item:IsA("Hat") or item:IsA("Accoutrement") or item:IsA("Shirt") or item:IsA("Pants") or item:IsA("ShirtGraphic") or item:IsA("BodyColors") or item:IsA("CharacterMesh") or item:IsA("WrapLayer") or item:IsA("Clothing") then
                item:Destroy()
            end
        end
        
        local targetHead = targetChar:FindFirstChild("Head")
        if targetHead and data.HeadBackup then
            for _, child in pairs(targetHead:GetChildren()) do
                child:Destroy()
            end
            
            targetHead.Transparency = data.HeadBackup.Transparency
            targetHead.Color = data.HeadBackup.Color
            
            if targetHead:IsA("MeshPart") then
                targetHead.MeshId = data.HeadBackup.MeshId or ""
                targetHead.TextureID = data.HeadBackup.TextureID or ""
                if data.HeadBackup.Size then
                    targetHead.Size = data.HeadBackup.Size
                end
            end
            
            for _, child in pairs(data.HeadBackup.Children) do
                child:Clone().Parent = targetHead
            end
        end
        
        for _, part in pairs(targetChar:GetChildren()) do
            if part:IsA("BasePart") then
                for _, child in pairs(part:GetChildren()) do
                    if child.Name == "VisualChangerWeld" then
                        child:Destroy()
                    end
                end
            end
        end
        
        for _, item in pairs(data.OriginalItems) do
            if item:IsA("Accessory") or item:IsA("Hat") or item:IsA("Accoutrement") then
                attachAccessoryToTarget(targetChar, item:Clone())
            else
                item:Clone().Parent = targetChar
            end
        end
        
        setCustomDisplayName(targetChar, data.OriginalName)
        modifiedPlayers[selectedPlayer] = nil
        
        RestoreBtn.Text = "Restaurado!"
        task.wait(1.5)
        RestoreBtn.Text = "Restaurar Original"
    end
end
