-- ================= TRANSPARENCIA (CACHE COMPLETO) ================= --

local function hideCharacterAndCache(char)
    local cache = {}
    for _, obj in ipairs(char:GetDescendants()) do
        if obj:IsA("BasePart") and obj.Name ~= "HumanoidRootPart" then
            cache[obj] = obj.Transparency
            obj.Transparency = 1
        elseif obj:IsA("Decal") or obj:IsA("Texture") then
            cache[obj] = obj.Transparency
            obj.Transparency = 1
        elseif obj:IsA("ParticleEmitter") or obj:IsA("Beam") or obj:IsA("Trail") then
            cache[obj] = obj.Enabled
            obj.Enabled = false
        elseif obj:IsA("BillboardGui") or obj:IsA("SurfaceGui") then
            cache[obj] = obj.Enabled
            obj.Enabled = false
        end
    end
    return cache
end

local function restoreCharacterVisibility(cache)
    if not cache then return end
    for obj, originalValue in pairs(cache) do
        if obj and obj.Parent then
            if obj:IsA("BillboardGui") or obj:IsA("SurfaceGui")
                or obj:IsA("ParticleEmitter") or obj:IsA("Beam") or obj:IsA("Trail") then
                obj.Enabled = originalValue
            else
                obj.Transparency = originalValue
            end
        end
    end
end

-- ================= ALTURA DINAMICA (resolve flutuacao) ================= --

-- Calcula o Y do ponto mais baixo (sola dos pes) relativo ao HumanoidRootPart.
-- Funciona para R6 e R15 de qualquer escala, inclusive avatares menores.
local function getBottomOffset(model, rootPart)
    local lowestY = 0
    for _, part in ipairs(model:GetDescendants()) do
        if part:IsA("BasePart") and part ~= rootPart then
            local relative = rootPart.CFrame:ToObjectSpace(part.CFrame)
            local bottom = relative.Position.Y - (part.Size.Y / 2)
            if bottom < lowestY then
                lowestY = bottom
            end
        end
    end
    return lowestY
end

-- ================= ANIMACOES DO ALVO (Korblox incluso) ================= --

-- Busca os IDs de animacao customizados do jogador (Korblox, pacotes, etc.)
local function fetchTargetAnimations(userId, rigType)
    local animations = {}
    local success, appearance = pcall(function()
        return Players:GetCharacterAppearanceAsync(userId)
    end)

    if success and appearance then
        appearance.Parent = game:GetService("ReplicatedStorage")

        local folderName = (rigType == Enum.HumanoidRigType.R15) and "R15Anim" or "R6Anim"
        local animFolder = appearance:FindFirstChild(folderName)

        if animFolder then
            for _, stateFolder in ipairs(animFolder:GetChildren()) do
                local ids = {}
                for _, anim in ipairs(stateFolder:GetChildren()) do
                    if anim:IsA("Animation") and anim.AnimationId ~= "" then
                        table.insert(ids, anim.AnimationId)
                    end
                end
                if #ids > 0 then
                    animations[stateFolder.Name:lower()] = ids
                end
            end
        end

        appearance:Destroy()
    end

    return animations
end

-- ================= MAPEAMENTO DE MOTORES (espelhamento confiavel) ================= --

-- Mapeia Motor6D pelo nome do Part1 (parte do corpo que ele controla).
-- Mais confiavel que FindFirstChild por nome, especialmente com acessorios.
local function buildMotorMap(model)
    local map = {}
    for _, motor in ipairs(model:GetDescendants()) do
        if motor:IsA("Motor6D") and motor.Part1 then
            map[motor.Part1.Name] = motor
        end
    end
    return map
end

-- ================= RESTAURAR ================= --

local function internalRestore(targetPlayer)
    if not targetPlayer or not modifiedPlayers[targetPlayer] then return end
    local data = modifiedPlayers[targetPlayer]

    if data.Connection then data.Connection:Disconnect() end
    if data.Dummy then data.Dummy:Destroy() end
    restoreCharacterVisibility(data.VisibilityCache)

    modifiedPlayers[targetPlayer] = nil
end

-- ================= SISTEMA PRINCIPAL ================= --

local function processCloning(mode)
    if not selectedPlayer then return end

    local targetChar = selectedPlayer.Character
    local targetHum = targetChar and targetChar:FindFirstChildOfClass("Humanoid")
    local realRoot = targetChar and targetChar:FindFirstChild("HumanoidRootPart")
    if not targetChar or not targetHum or not realRoot then return end

    local copyUsername = UsernameInput.Text
    if copyUsername == "" then copyUsername = selectedPlayer.Name end

    local btn = (mode == "R15") and ApplyR15Btn or ApplyR6Btn
    local defaultText = btn.Text
    btn.Text = "Carregando..."

    internalRestore(selectedPlayer)

    local success, targetId = pcall(function() return Players:GetUserIdFromNameAsync(copyUsername) end)
    if not success or not targetId then
        btn.Text = "Usuario Invalido!"
        task.delay(3, function() btn.Text = defaultText end)
        return
    end

    local rigType = (mode == "R15") and Enum.HumanoidRigType.R15 or Enum.HumanoidRigType.R6
    local modelSuccess, sourceModel = pcall(function()
        return Players:CreateHumanoidModelFromUserId(targetId, rigType)
    end)

    if modelSuccess and sourceModel then
        -- Esconde o personagem original e salva o cache
        local visCache = hideCharacterAndCache(targetChar)

        local dummy = sourceModel
        dummy.Name = "VisualCloneDummy_" .. copyUsername

        local fakeHum = dummy:FindFirstChildOfClass("Humanoid")
        if fakeHum then
            fakeHum.DisplayDistanceType = Enum.HumanoidDisplayDistanceType.None
            fakeHum.HealthDisplayType = Enum.HumanoidHealthDisplayType.AlwaysOff
            fakeHum.PlatformStand = true
            fakeHum.AutoRotate = false
            fakeHum.WalkSpeed = 0
            fakeHum.JumpPower = 0
            fakeHum.JumpHeight = 0
        end

        -- Remove fisica e scripts do dummy, mas PRESERVA tudo que e visual
        -- (MeshParts, efeitos, particles, etc.) para o Korblox funcionar.
        for _, part in ipairs(dummy:GetDescendants()) do
            if part:IsA("BasePart") then
                part.CanCollide = false
                part.Massless = true
                part.Anchored = false
                part.CanTouch = false
                part.CanQuery = false
            elseif part:IsA("LocalScript") then
                -- Remove LocalScripts (poderiam conflitar com o jogo),
                -- mas mantem Scripts comuns caso o bundle precise deles.
                part:Destroy()
            end
        end

        dummy.Parent = targetChar

        local fakeRoot = dummy:FindFirstChild("HumanoidRootPart")
        local connection

        if fakeRoot then
            -- Ancora temporariamente para o dummy nao cair antes do weld
            fakeRoot.Anchored = true

            -- === ALTURA DINAMICA ===
            -- Alinha a sola dos pes dos dois rigs, independente de escala.
            local realBottom = getBottomOffset(targetChar, realRoot)
            local fakeBottom = getBottomOffset(dummy, fakeRoot)
            local yAlign = realBottom - fakeBottom

            fakeRoot.CFrame = realRoot.CFrame * CFrame.new(0, yAlign, 0)

            local weld = Instance.new("WeldConstraint")
            weld.Part0 = realRoot
            weld.Part1 = fakeRoot
            weld.Parent = fakeRoot

            fakeRoot.Anchored = false

            local sameRig = (targetHum.RigType == rigType)

            if sameRig then
                -- ========================================================
                -- MESMO RIG (R6->R6 ou R15->R15): espelha Motor6D
                -- ========================================================

                -- Remove o Animator do dummy para que a maquina de estados
                -- nao sobrescreva nossos transforms com poses padrao.
                if fakeHum then
                    local animator = fakeHum:FindFirstChildOfClass("Animator")
                    if animator then animator:Destroy() end
                end

                -- Mapeia os motores do dummy pelo nome da parte do corpo
                local fakeMotors = buildMotorMap(dummy)

                -- IMPORTANTE: usa Heartbeat, NAO Stepped.
                -- Stepped roda ANTES da atualizacao de animacao -> transforms
                -- eram sobrescritos (era esse o bug da R6 sem animacao).
                -- Heartbeat roda DEPOIS -> nossos transforms sao os finais.
                connection = RunService.Heartbeat:Connect(function()
                    if not targetChar.Parent or not dummy.Parent then return end

                    for _, realMotor in ipairs(targetChar:GetDescendants()) do
                        -- Pula motores que pertencem ao proprio dummy
                        if realMotor:IsA("Motor6D") and not realMotor:IsDescendantOf(dummy) then
                            local partName = realMotor.Part1 and realMotor.Part1.Name
                            local fakeMotor = partName and fakeMotors[partName]
                            if fakeMotor then
                                fakeMotor.Transform = realMotor.Transform
                            end
                        end
                    end
                end)

            else
                -- ========================================================
                -- RIG DIFERENTE (R15->R6 ou R6->R15): toca animacoes
                -- ========================================================

                -- Garante um Animator no dummy
                local animator = fakeHum and fakeHum:FindFirstChildOfClass("Animator")
                if not animator and fakeHum then
                    animator = Instance.new("Animator")
                    animator.Parent = fakeHum
                end

                -- Busca animacoes customizadas do alvo (Korblox, etc.)
                local customAnims = fetchTargetAnimations(targetId, rigType)

                -- Usa animacao customizada se existir; senao, fallback
                local function getAnimId(state, fallbackId)
                    local ids = customAnims[state:lower()]
                    if ids and #ids > 0 then
                        return ids[1]
                    end
                    return "rbxassetid://" .. fallbackId
                end

                local animIDs
                if mode == "R6" then
                    animIDs = {
                        Idle     = getAnimId("idle",     "180435571"),
                        Walk     = getAnimId("walk",     "180426354"),
                        Run      = getAnimId("run",      "180426354"),
                        Jump     = getAnimId("jump",     "125750702"),
                        Fall     = getAnimId("fall",     "180436148"),
                        Climb    = getAnimId("climb",    "180436334"),
                        Swim     = getAnimId("swim",     "180435571"),
                        SwimIdle = getAnimId("swimidle", "180435571"),
                    }
                else
                    animIDs = {
                        Idle     = getAnimId("idle",     "507766666"),
                        Walk     = getAnimId("walk",     "507777826"),
                        Run      = getAnimId("run",      "507777826"),
                        Jump     = getAnimId("jump",     "507765000"),
                        Fall     = getAnimId("fall",     "507767968"),
                        Climb    = getAnimId("climb",    "507765644"),
                        Swim     = getAnimId("swim",     "507770818"),
                        SwimIdle = getAnimId("swimidle", "507770239"),
                    }
                end

                local loaded = {}
                for name, id in pairs(animIDs) do
                    local a = Instance.new("Animation")
                    a.AnimationId = id
                    local track = animator:LoadAnimation(a)
                    track.Priority = Enum.AnimationPriority.Movement
                    loaded[name] = track
                end

                local currentAnim = "Idle"
                loaded.Idle:Play()

                connection = RunService.RenderStepped:Connect(function()
                    if not targetChar.Parent or not targetHum or not realRoot
                        or not dummy or not dummy.Parent then return end

                    local state = targetHum:GetState()
                    local speed = (realRoot.Velocity * Vector3.new(1, 0, 1)).Magnitude
                    local desiredAnim = "Idle"

                    if state == Enum.HumanoidStateType.Swimming then
                        desiredAnim = (speed > 1) and "Swim" or "SwimIdle"
                    elseif state == Enum.HumanoidStateType.Climbing then
                        desiredAnim = "Climb"
                    elseif state == Enum.HumanoidStateType.Freefall then
                        desiredAnim = "Fall"
                    elseif state == Enum.HumanoidStateType.Jumping then
                        desiredAnim = "Jump"
                    elseif speed > 1 then
                        desiredAnim = "Walk"
                    end

                    if currentAnim ~= desiredAnim then
                        if loaded[currentAnim] then loaded[currentAnim]:Stop(0.2) end
                        if loaded[desiredAnim] then loaded[desiredAnim]:Play(0.2) end
                        currentAnim = desiredAnim
                    end

                    if currentAnim == "Walk" or currentAnim == "Run" then
                        loaded[currentAnim]:AdjustSpeed(speed / 16)
                    end
                end)
            end

            modifiedPlayers[selectedPlayer] = {
                Dummy = dummy,
                Connection = connection,
                VisibilityCache = visCache,
            }
            btn.Text = "Sucesso!"
        else
            btn.Text = "Erro: Sem Root!"
        end
    else
        btn.Text = "Erro ao Baixar!"
    end

    task.delay(3, function() btn.Text = defaultText end)
end

-- ================= EVENTOS DA GUI ================= --
local function updateDropdown()
    for _, child in pairs(DropdownScroll:GetChildren()) do
        if child:IsA("TextButton") then child:Destroy() end
    end
    local ySize = 0
    for _, player in pairs(Players:GetPlayers()) do
        local btn = Instance.new("TextButton")
        btn.Size = UDim2.new(1, 0, 0, 30)
        btn.BackgroundColor3 = Color3.fromRGB(20, 20, 20)
        btn.BackgroundTransparency = 0.2
        btn.TextColor3 = Color3.fromRGB(255, 255, 255)
        btn.Text = player.Name .. (player == LocalPlayer and " (Voce)" or "")
        btn.Font = Enum.Font.Gotham
        btn.TextSize = 12
        btn.ZIndex = 16
        btn.Parent = DropdownScroll
        Instance.new("UICorner", btn).CornerRadius = UDim.new(0, 4)
        btn.Activated:Connect(function()
            selectedPlayer = player
            DropdownBtn.Text = "Alvo: " .. player.Name
            DropdownScroll.Visible = false
            DropdownBackdrop.Visible = false
        end)
        ySize = ySize + 30
    end
    DropdownScroll.CanvasSize = UDim2.new(0, 0, 0, ySize)
end

-- Limpa dummy quando o personagem respawna
local function hookRespawn(player)
    player.CharacterAdded:Connect(function()
        if modifiedPlayers[player] then
            internalRestore(player)
        end
    end)
end

    hookRespawn(p)
    updateDropdown()
end)

    DropdownScroll.Visible = not DropdownScroll.Visible
    DropdownBackdrop.Visible = DropdownScroll.Visible
